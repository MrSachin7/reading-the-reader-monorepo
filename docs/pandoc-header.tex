% Cap embedded screenshots so tall ones leave room for their caption and float
% spacing (prevents "Float too large for page" overfull warnings). This is a
% faithful copy of pandoc's own \pandocbounded, with the height budget reduced
% from \textheight to 0.86\textheight. Runs after the template defines the macro.
\makeatletter
\renewcommand*\pandocbounded[1]{%
  \sbox\pandoc@box{#1}%
  \Gscale@div\@tempa{\dimexpr0.86\textheight\relax}{\dimexpr\ht\pandoc@box+\dp\pandoc@box\relax}%
  \Gscale@div\@tempb{\linewidth}{\wd\pandoc@box}%
  \ifdim\@tempb\p@<\@tempa\p@\let\@tempa\@tempb\fi%
  \ifdim\@tempa\p@<\p@\scalebox{\@tempa}{\usebox\pandoc@box}%
  \else\usebox{\pandoc@box}%
  \fi%
}
\makeatother
